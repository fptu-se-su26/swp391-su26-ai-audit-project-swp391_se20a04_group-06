export interface ProductProps {

  name: string;

  description: string;

  price: number;

  salesType: "Retail" | "Wholesale";

  totalWeight: number;
  // Khối lượng còn lại của sản phẩm sau khi đã bán (đơn vị kg)
  remainingWeight: number;
  // Trạng thái hiện tại của sản phẩm, gồm: Active (Đang bán), SoldOut (Hết hàng), Expired (Hết hạn), Deleted (Đã xóa)
  status: "Active" | "SoldOut" | "Expired" | "Deleted";

  // Hạn sử dụng của sản phẩm (nếu có)

  images: string[];
  // Thời điểm sản phẩm được đẩy bài viết (bump) gần nhất để xếp lên đầu danh sách tìm kiếm
  bumpedAt?: Date;
